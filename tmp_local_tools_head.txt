147: function resolvePath(p: string): string {
148:   if (path.isAbsolute(p)) return p;
149:   const ws = vscode.workspace.workspaceFolders?.[0]?.uri.fsPath;
150:   return ws ? path.resolve(ws, p) : p;
151: }

162: /*  Per-file Mutex — prevents concurrent edits to the same file       */
163: /* ------------------------------------------------------------------ */
164: 
165: /**
166:  * Simple async mutex keyed by file path. When Anthropic models fire
167:  * multiple modify_entity / write_file calls targeting the same file
168:  * simultaneously, the mutex serialises them so only one edit is
169:  * in-flight per file at a time. Different files can still be edited
170:  * concurrently without blocking.
171:  */
172: const fileLocks = new Map<string, Promise<void>>();
173: 
174: async function withFileLock<T>(absPath: string, fn: () => Promise<T>): Promise<T> {
175:   const key = absPath.toLowerCase(); // normalise on Windows
176:   const prev = fileLocks.get(key) ?? Promise.resolve();
177:   let release!: () => void;
178:   const next = new Promise<void>((r) => { release = r; });
179:   fileLocks.set(key, next);
180:   try {
181:     await prev; // wait for previous edit on this file to finish
182:     return await fn();
183:   } finally {
184:     release();
185:     // GC: remove the entry only if we're still the tail
186:     if (fileLocks.get(key) === next) fileLocks.delete(key);
187:   }
188: }

194: /** Rough count of top-level members inside a class/interface body. */
195: function countMembers(text: string): number {
196:   // Heuristic: count lines that start a method, property, or constructor
197:   const memberPattern = /^\s+(public|private|protected|static|readonly|abstract|async|get |set |constructor\b|\w+\s*\()/gm;
198:   const matches = text.match(memberPattern);
199:   return matches?.length ?? 0;
200: }
